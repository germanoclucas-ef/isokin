async function generateAiReport() { const apiKey = localStorage.getItem('isokin_gemini_key'); const model = localStorage.getItem('isokin_gemini_model') || 'gemini-2.5-flash'; if (!apiKey) { alert('Configure sua chave de API no botão "AI Studio API" no topo antes de gerar.'); openAiModal(); return; } const btn = document.getElementById('btn-ai-gen'); btn.innerText = 'Processando com Gemini...'; btn.disabled = true; const payload = { aluno: document.getElementById('eval-name').value || 'Não informado', massaCorporal: document.getElementById('eval-weight').value || 'Não informado', modalidade: document.getElementById('eval-modality').value, contextoClinico: document.getElementById('eval-context').value, joelho: { isquiosE: document.getElementById('ham-l-peak').innerText, isquiosD: document.getElementById('ham-r-peak').innerText, quadE: document.getElementById('quad-l-peak').innerText, quadD: document.getElementById('quad-r-peak').innerText }, cotovelo: { bicepsE: document.getElementById('biceps-l-peak').innerText, bicepsD: document.getElementById('biceps-r-peak').innerText, tricepsE: document.getElementById('triceps-l-peak').innerText, tricepsD: document.getElementById('triceps-r-peak').innerText }, padraoHorizontal: { supinoE: document.getElementById('press-l-peak').innerText, supinoD: document.getElementById('press-r-peak').innerText, remadaE: document.getElementById('row-l-peak').innerText, remadaD: document.getElementById('row-r-peak').innerText }, padraoVertical: { desenvolvimentoE: document.getElementById('ohp-l-peak').innerText, desenvolvimentoD: document.getElementById('ohp-r-peak').innerText, puxadorE: document.getElementById('pull-l-peak').innerText, puxadorD: document.getElementById('pull-r-peak').innerText }, corePallof: { esquerdo: document.getElementById('pallof-l-peak').innerText, direito: document.getElementById('pallof-r-peak').innerText } }; const promptText = `Você é um especialista em biomecânica e avaliação neuromuscular com balança de tração isométrica. Analise os dados da bateria abaixo (incluindo membros superiores, padrões horizontais/verticais e estabilidade de core anti-rotacional) e emita um parecer clínico e prescritivo fundamentado:\n\n${JSON.stringify(payload, null, 2)}`; try { const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: promptText }] }] }) }); const data = await response.json(); if (data.error) { alert('Detalhe do erro retornado pela Google: ' + data.error.message); return; } if (data.candidates && data.candidates[0].content && data.candidates[0].content.parts[0].text) { document.getElementById('diagnostic-text').innerHTML = `
Parecer Gerado via Gemini (${model}):
${data.candidates[0].content.parts[0].text}
`; } else { alert('Resposta incompleta da API: ' + JSON.stringify(data)); } } catch (err) { alert('Falha na comunicação de rede com o Google AI Studio: ' + err.message); } finally { btn.innerText = 'Gerar Parecer via Google AI Studio'; btn.disabled = false; } }
